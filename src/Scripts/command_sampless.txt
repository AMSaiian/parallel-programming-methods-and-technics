# Sample commands for all Runner scenarios
# Run from: src/Runner/
# Usage:  dotnet run <command> [options]
#
# Global options (available on all commands):
#   -t, --threads <n>   (REQUIRED) Number of threads
#   -s, --seed <n>      Random seed for reproducibility (default: Environment.TickCount)
#   -vs, --verbose      Show detailed result output (flag, no value needed)

# ---------------------------------------------------------------------------
# monte  –  MonteCarlo π estimation
#   -i, --iterations <n>   (required) Number of random samples
# ---------------------------------------------------------------------------

dotnet run monte -t 2  -i 10000000 -s 42 -vs
dotnet run monte -t 4  -i 10000000 -s 42 -vs
dotnet run monte -t 6  -i 10000000 -s 42 -vs
dotnet run monte -t 12 -i 10000000 -s 42 -vs

# ---------------------------------------------------------------------------
# factor  –  Factorial computation (BigInteger, computes 1! through n!)
#   -m, --member <n>   (required) Compute up to n!
# ---------------------------------------------------------------------------

dotnet run factor -t 2  -m 10000
dotnet run factor -t 4  -m 10000
dotnet run factor -t 6  -m 10000
dotnet run factor -t 12 -m 10000

# ---------------------------------------------------------------------------
# simple  –  Prime number search (trial division)
#   -lo, --lower <n>   (required) Lower bound (inclusive, >= 2)
#   -up, --upper <n>   (required) Upper bound (inclusive)
# ---------------------------------------------------------------------------

dotnet run simple -t 2  -lo 2 -up 10000000 -vs
dotnet run simple -t 4  -lo 2 -up 10000000 -vs
dotnet run simple -t 6  -lo 2 -up 10000000 -vs
dotnet run simple -t 12 -lo 2 -up 10000000 -vs

# ---------------------------------------------------------------------------
# matrix  –  N×N matrix transposition (double[,])
#   -d, --dim <n>   Matrix dimension N (default: 10000)
# ---------------------------------------------------------------------------

dotnet run matrix -t 2  -d 10000 -s 42 -vs
dotnet run matrix -t 4  -d 10000 -s 42 -vs
dotnet run matrix -t 6  -d 10000 -s 42 -vs
dotnet run matrix -t 12 -d 10000 -s 42 -vs

# ---------------------------------------------------------------------------
# words  –  Recursive word count in randomly generated text files
#   -fc, --files <n>     Number of .txt files to generate (1–10000, default: 1000)
#   -w,  --words <n>     Max words per file, uniform [0, n] (1–100000, default: 500)
#   -fd, --folders <n>   Number of subdirectories in tree (1–10000, default: 100)
#   -td, --teardown      Preserve generated files after run (flag, default: delete)
# ---------------------------------------------------------------------------

dotnet run words -t 2  -fc 10000 -w 10000 -fd 1000 -s 42 -vs
dotnet run words -t 4  -fc 10000 -w 10000 -fd 1000 -s 42 -vs
dotnet run words -t 6  -fc 10000 -w 10000 -fd 1000 -s 42 -vs
dotnet run words -t 12 -fc 10000 -w 10000 -fd 1000 -s 42 -vs

# ---------------------------------------------------------------------------
# stats  –  Compute min, max, mean and median of a large random array
#   -al,  --algo <name>   (required) Pattern: sequential | reducemap | forkjoin | workerpool
#   -sz, --size <n>      Number of elements (>= 1 000 000, default: 1 000 000)
#   -sq, --seq           Also run sequential and print speedup + efficiency (flag)
# ---------------------------------------------------------------------------

dotnet run stats -t 2  -al reducemap   -sz 5000000 -s 42 -sq
dotnet run stats -t 4  -al reducemap   -sz 5000000 -s 42 -sq
dotnet run stats -t 6  -al reducemap   -sz 5000000 -s 42 -sq
dotnet run stats -t 12 -al reducemap   -sz 5000000 -s 42 -sq

dotnet run stats -t 2  -al forkjoin    -sz 5000000 -s 42 -sq
dotnet run stats -t 4  -al forkjoin    -sz 5000000 -s 42 -sq
dotnet run stats -t 6  -al forkjoin    -sz 5000000 -s 42 -sq
dotnet run stats -t 12 -al forkjoin    -sz 5000000 -s 42 -sq

dotnet run stats -t 2  -al workerpool  -sz 5000000 -s 42 -sq
dotnet run stats -t 4  -al workerpool  -sz 5000000 -s 42 -sq
dotnet run stats -t 6  -al workerpool  -sz 5000000 -s 42 -sq
dotnet run stats -t 12 -al workerpool  -sz 5000000 -s 42 -sq

# ---------------------------------------------------------------------------
# tags  –  Count HTML tag frequencies across a dataset of HTML files
#   -al,  --algo <name>   (required) Pattern: sequential | reducemap | forkjoin | workerpool

# ---------------------------------------------------------------------------
# transactions  –  Financial transaction processing: currency conversion, cashback, aggregation
#   -al,  --algo <name>   (required) Pattern: sequential | pipeline | producerconsumer
#   -c,  --count <n>      Number of transactions to generate (>= 10 000, default: 100 000)
#   -sq, --seq            Also run sequential and print speedup + efficiency (flag)
#   -vs, --verbose        Print first 5 transactions with per-step amounts
# ---------------------------------------------------------------------------

dotnet run transactions -t 2  -al pipeline          -ct 500000 -s 42 -sq
dotnet run transactions -t 4  -al pipeline          -ct 500000 -s 42 -sq
dotnet run transactions -t 6  -al pipeline          -ct 500000 -s 42 -sq
dotnet run transactions -t 12 -al pipeline          -ct 500000 -s 42 -sq

dotnet run transactions -t 4  -al producerconsumer  -ct 500000 -s 42 -sq
dotnet run transactions -t 6  -al producerconsumer  -ct 500000 -s 42 -sq
dotnet run transactions -t 12 -al producerconsumer  -ct 500000 -s 42 -sq

# ---------------------------------------------------------------------------
# bank-transfer  –  Bank account transfers: race condition, deadlock, safe sync
#   -sc, --scenario <name>   (required) Scenario: race | deadlock | safe
#   -t,  --threads <n>       (required) Thread pool size
#   -w,  --workers <n>       Number of concurrent worker tasks (>= 1000, default: 1000)
#   -ac, --accounts <n>      Number of accounts (>= 100, default: 200)
#   -d,  --duration <n>      Run duration in seconds (default: 5)
#   -vs, --verbose           Print sample account balances after run
# ---------------------------------------------------------------------------

dotnet run bank-transfer -t 12 -w 1000 -sc race     -ac 200 -d 5 -s 42 -vs
dotnet run bank-transfer -t 12 -w 1000 -sc deadlock -ac 200 -d 5 -s 42 -vs
dotnet run bank-transfer -t 12 -w 1000 -sc safe     -ac 200 -d 5 -s 42 -vs

# ---------------------------------------------------------------------------
# ipc-demo  –  IPC between C# (main) and Java/MPJ Express (supplementary)
#   Run via Scripts/run-ipc-demo.ps1 which starts both sides simultaneously.
#
#   run-ipc-demo.ps1 options:
#   -Method <name>     (required) sharedmemory | namedpipe | tcp
#   -Threads <n>       C# thread pool size (default: 12)
#   -Seed <n>          Random seed (default: 42)
#   -MpjHome <path>    MPJ Express dir (default: C:\mpj-v0_44)
#   -Loud           Print verbose output
#
# Prerequisites:
#   • JDK on PATH (javac + java)
#   • MPJ Express installed at C:\mpj-v0_44 (or pass -MpjHome)
# ---------------------------------------------------------------------------

.\Scripts\run-ipc-demo.ps1 -Method sharedmemory -Loud
.\Scripts\run-ipc-demo.ps1 -Method namedpipe    -Loud
.\Scripts\run-ipc-demo.ps1 -Method tcp          -Loud
